```typescript
// src/config/weather.ts
export type WeatherRarity = 'common' | 'uncommon' | 'rare' | 'very_rare'

export type WeatherTypeConfig = {
  id: string
  name: string
  metalMultiplier: number
  gasMultiplier: number
  energyMultiplier: number
  durationMinHours: number
  durationMaxHours: number
  rarity: WeatherRarity
  weight: number
}

export const WEATHER_TYPES: WeatherTypeConfig[] = [
  { id: 'calm_skies', name: 'Calm Skies', metalMultiplier: 1.0, gasMultiplier: 1.0, energyMultiplier: 1.0, durationMinHours: 0, durationMaxHours: 0, rarity: 'common', weight: 26.5 },
  { id: 'solar_flare', name: 'Solar Flare', metalMultiplier: 1.0, gasMultiplier: 1.0, energyMultiplier: 1.3, durationMinHours: 2, durationMaxHours: 3, rarity: 'common', weight: 26.5 },
  { id: 'metal_vein', name: 'Metal Vein Discovered', metalMultiplier: 1.25, gasMultiplier: 1.0, energyMultiplier: 1.0, durationMinHours: 3, durationMaxHours: 4, rarity: 'uncommon', weight: 10 },
  { id: 'gas_pocket', name: 'Gas Pocket', metalMultiplier: 1.0, gasMultiplier: 1.25, energyMultiplier: 1.0, durationMinHours: 3, durationMaxHours: 4, rarity: 'uncommon', weight: 10 },
  { id: 'ion_storm', name: 'Ion Storm', metalMultiplier: 1.0, gasMultiplier: 0.8, energyMultiplier: 0.6, durationMinHours: 2, durationMaxHours: 3, rarity: 'uncommon', weight: 10 },
  { id: 'dust_storm', name: 'Dust Storm', metalMultiplier: 0.7, gasMultiplier: 1.0, energyMultiplier: 0.85, durationMinHours: 2, durationMaxHours: 4, rarity: 'rare', weight: 4 },
  { id: 'solar_storm', name: 'Solar Storm', metalMultiplier: 0.5, gasMultiplier: 0.5, energyMultiplier: 0.4, durationMinHours: 1, durationMaxHours: 2, rarity: 'very_rare', weight: 5 },
  { id: 'asteroid_shower', name: 'Asteroid Shower', metalMultiplier: 1.8, gasMultiplier: 0.6, energyMultiplier: 0.9, durationMinHours: 1, durationMaxHours: 1, rarity: 'rare', weight: 4 },
  { id: 'nebula_drift', name: 'Nebula Drift', metalMultiplier: 0.9, gasMultiplier: 1.8, energyMultiplier: 1.1, durationMinHours: 1, durationMaxHours: 1, rarity: 'rare', weight: 4 },
]

const WEATHER_MAP = new Map(WEATHER_TYPES.map((w) => [w.id, w]))

export function getWeatherConfig(id: string): WeatherTypeConfig {
  const config = WEATHER_MAP.get(id)
  if (!config) throw new Error(`Unknown weather type: ${id}`)
  return config
}

/** Roll a random weather type based on rarity weights. Pass a value 0–1 for deterministic testing. */
export function rollWeatherType(randomValue?: number): string {
  const totalWeight = WEATHER_TYPES.reduce((sum, w) => sum + w.weight, 0)
  let roll = (randomValue ?? Math.random()) * totalWeight
  for (const w of WEATHER_TYPES) {
    roll -= w.weight
    if (roll <= 0) return w.id
  }
  return WEATHER_TYPES[0].id
}

/** Roll a random duration in hours within the weather type's range. */
export function rollWeatherDurationHours(id: string): number {
  const config = getWeatherConfig(id)
  if (config.durationMinHours === 0 && config.durationMaxHours === 0) return 0
  return config.durationMinHours + Math.random() * (config.durationMaxHours - config.durationMinHours)
}
```

```typescript
// src/hooks/useResearch.ts
import { useState, useEffect, useCallback } from 'react'
import { supabase } from '../lib/supabase'
import { IS_DEV_MODE } from '../lib/devMode'
import type { ResearchQueueItem } from './usePlanet'

export function useResearch(
  planetId: string | undefined,
  researchQueue: ResearchQueueItem[],
  onComplete: () => void
) {
  const [timeRemaining, setTimeRemaining] = useState<number | null>(null)

  const activeResearch = researchQueue.length > 0 ? researchQueue[0] : null

  useEffect(() => {
    if (!activeResearch) {
      setTimeRemaining(null)
      return
    }

    const update = () => {
      const ms = new Date(activeResearch.completes_at).getTime() - Date.now()
      setTimeRemaining(Math.max(0, ms))
      if (ms <= 0) {
        completeResearch()
      }
    }

    const completeResearch = async () => {
      if (!planetId) return
      await supabase.functions.invoke('game-action', {
        body: { action: 'complete_research', planetId },
      })
      onComplete()
    }

    update()
    const interval = setInterval(update, 1000)
    return () => clearInterval(interval)
  }, [activeResearch, planetId, onComplete])

  const startResearch = useCallback(
    async (techId: string) => {
      if (!planetId) return
      const { data, error } = await supabase.functions.invoke('game-action', {
        body: { action: 'start_research', planetId, techId, devMode: IS_DEV_MODE },
      })
      if (error) throw error
      if (data?.error) throw new Error(data.error)
      onComplete()
      return data
    },
    [planetId, onComplete]
  )

  return { activeResearch, researchTimeRemaining: timeRemaining, startResearch }
}
```

```typescript
// deno-lint-ignore no-explicit-any
async function detectRadarCoordinates(supabase: any, planetId: string, radarLevel: number) {
  const { data: planet } = await supabase
    .from('planets').select('coordinates').eq('id', planetId).single()

  if (!planet) return

  const home = parseCoord(planet.coordinates)
  const range = radarLevel * RADAR_RANGE_PER_LEVEL // systems in each direction

  const newDetections: { planet_id: string; coordinates: string; visibility: string }[] = []

  for (let sys = Math.max(1, home.system - range); sys <= Math.min(500, home.system + range); sys++) {
    for (let pos = 1; pos <= 15; pos++) {
      const coords = `${home.galaxy}:${sys}:${pos}`
      if (coords === planet.coordinates) continue // skip home
      newDetections.push({ planet_id: planetId, coordinates: coords, visibility: 'detected' })
    }
  }

  if (newDetections.length > 0) {
    // Upsert — only insert if not already existing (don't downgrade revealed to detected)
    await supabase
      .from('galaxy_map')
      .upsert(newDetections, { onConflict: 'planet_id,coordinates', ignoreDuplicates: true })

    await supabase.from('planet_events').insert({
      planet_id: planetId,
      event_type: 'radar_detection',
      message: `Radar Array upgraded — new signals detected across ${range * 2 + 1} systems`,
      metadata: { radar_level: radarLevel, systems_in_range: range * 2 + 1 },
    })
  }
}
```

```typescript
  const now = new Date()

  // Find planets where current weather has expired
  const { data: expiredWeather } = await supabase
    .from('planet_weather')
    .select('planet_id')
    .not('expires_at', 'is', null)
    .lte('expires_at', now.toISOString())

  // Find planets with no active weather (expires_at is null = permanent calm_skies from init)
  // We want to start rotating weather for those too
  const { data: permanentWeather } = await supabase
    .from('planet_weather')
    .select('planet_id, started_at')
    .is('expires_at', null)

  const planetIds = new Set<string>()
  for (const row of expiredWeather ?? []) planetIds.add(row.planet_id)
  // Only rotate permanent weather if it was set more than 4 hours ago
  for (const row of permanentWeather ?? []) {
    const age = now.getTime() - new Date(row.started_at).getTime()
    if (age > 4 * 3600 * 1000) planetIds.add(row.planet_id)
  }

  let updated = 0

  for (const planetId of planetIds) {
    const weatherId = rollWeather()
    const config = WEATHER_TYPES[weatherId]

    const durationHours = config.minHours + Math.random() * (config.maxHours - config.minHours)
    const expiresAt = weatherId === 'calm_skies'
      ? new Date(now.getTime() + (4 + Math.random() * 2) * 3600 * 1000) // calm skies lasts 4-6 hours
      : new Date(now.getTime() + durationHours * 3600 * 1000)

    await supabase.from('planet_weather').insert({
      planet_id: planetId,
      weather_type: weatherId,
      metal_multiplier: config.metalMult,
      gas_multiplier: config.gasMult,
      energy_multiplier: config.energyMult,
      started_at: now.toISOString(),
      expires_at: expiresAt.toISOString(),
    })

    if (weatherId !== 'calm_skies') {
      await supabase.from('planet_events').insert({
        planet_id: planetId,
        event_type: 'weather_change',
        message: `Weather alert: ${weatherId.replace(/_/g, ' ')} incoming!`,
        metadata: { weather_type: weatherId },
      })
    }

    updated++
  }

  return new Response(JSON.stringify({ success: true, planetsUpdated: updated }), {
    headers: { 'Content-Type': 'application/json' },
  })
})
```

```typescript
  const seconds = Math.floor((Date.now() - date.getTime()) / 1000)

  if (minutes < 60) return `${minutes}m ago`
  const hours = Math.floor(minutes / 60)

  const days = Math.floor(hours / 24)
  return `${days}d ago`
}
```
